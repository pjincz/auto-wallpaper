cmake_minimum_required(VERSION 3.20)
project(AutoWallpaper LANGUAGES CXX RC)
add_executable(AutoWallpaper WIN32 src/main.cpp src/app.rc)
target_compile_features(AutoWallpaper PRIVATE cxx_std_17)
target_include_directories(AutoWallpaper PRIVATE src)
target_compile_definitions(AutoWallpaper PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(AutoWallpaper PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/EHsc>")
set_property(TARGET AutoWallpaper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(AutoWallpaper PRIVATE user32 shell32 ole32 winhttp windowsapp windowscodecs)
